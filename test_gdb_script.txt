b preprocessor_next_token
run
while 1
  finish
  if $rax != 0
    printf "preprocessor_next_token returned error: %d\n", $rax
    set $pr = $rbx
    set $lx = *(void**)($pr + 48)
    printf "File: %s, Line: %d, Col: %d\n", *(char**)($lx + 32), *(int*)($lx + 40), *(short*)($lx + 44)
    bt
    quit
  end
  continue
end
